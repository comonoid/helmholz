# --- переиспользование double-скретча как int32-массива ---
# fc->wbuf — скретч, выделенный под double (8 Б), в него кладут int32 (4 Б):
# «n int32 ≤ n double — места хватает». Буфер получен malloc/realloc, то есть не
# имеет объявленного типа; выравнивание double строже int32, поэтому на x86-64
# приведение корректно. На платформах со строгим выравниванием int32 это
# пришлось бы переписать (порт на GPU — отдельная задача, §882/HBLK).
src/nstruct/sweep.c	S2	invalidPointerCast	Casting from double * to signed int * is not portable due to different binary data representations on different platforms.	ОСОЗНАННО: скретч под double используется как int32 (выравнивание double строже); переписать при порте
